# RunPodFarm v2 — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Замкнуть цепочку «Cook в Houdini → зависимости на volume → GPU-поды считают → результаты в локальные пути ROP → поды погашены», переиспользуя всё рабочее из v1 (образ пода, Houdini с volume, лицензии, каркас HDA-шедулера, компрессия, cost tracker) и заменяя только транспорт (Redis/B2/JuiceFS → sftp на CPU sync-под + HTTP-воркер + PDG MQ на ферме).

**Architecture:** Артист-сторона = Python-пакет `rpfarm/` (общий для HDA и CLI) + 4 TOP-ноды. Ферма = один Network Volume с зонами, CPU sync-под (sshd, rclone, worker.py, mqserver) и GPU-поды (worker.py, hython с volume). Транспорт файлов — rclone по sftp на sync-под; управление — HTTPS к worker.py через RunPod proxy; результаты PDG — штатный MQ на sync-поде.

**Tech Stack:** Python 3.12/3.13 (Houdini 22 использует python3.13libs; на поде Ubuntu 24.04 python3.12), stdlib `urllib`/`http.server`, rclone (статический бинарь), PDG `PyScheduler` + `SimpleMQSchedulerMixin`, RunPod REST v1 (`https://rest.runpod.io/v1`), pytest.

**Spec:** `docs/superpowers/specs/2026-09-02-rpfarm-v2-design.md`

```python
# rpfarm/runpod_api.py
from __future__ import annotations
import json, urllib.request, urllib.error

BASE = "https://rest.runpod.io/v1"

class RunPodError(Exception):
    def __init__(self, status, body): super().__init__(f"RunPod {status}: {body[:300]}"); self.status, self.body = status, body

def _urllib_transport(method, url, body, headers):
    data = json.dumps(body).encode() if body is not None else None
    req = urllib.request.Request(url, method=method, data=data, headers=headers)
    try:
        with urllib.request.urlopen(req, timeout=60) as r:
            return r.status, r.read()
    except urllib.error.HTTPError as e:
        return e.code, e.read()

class RunPodAPI:
    def __init__(self, api_key, base_url=BASE, transport=_urllib_transport):
        self._h = {"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"}
        self.base, self._t = base_url.rstrip("/"), transport

    def _call(self, method, path, body=None, ok404=False):
        status, raw = self._t(method, self.base + path, body, self._h)
        if status == 404 and ok404: return None
        if status >= 300: raise RunPodError(status, raw.decode(errors="replace"))
        return json.loads(raw) if raw else None

    def _pod_body(self, name, template_id, volume_id, env, ports):
        return {"name": name, "templateId": template_id, "networkVolumeId": volume_id,
                "volumeMountPath": "/workspace", "env": env, "ports": ports,
                "cloudType": "SECURE", "supportPublicIp": True, "dataCenterIds": ["EU-RO-1"]}

    def create_gpu_pod(self, name, template_id, gpu_type_ids, volume_id, env, ports):
        b = self._pod_body(name, template_id, volume_id, env, ports)
        b.update({"computeType": "GPU", "gpuTypeIds": list(gpu_type_ids), "gpuTypePriority": "availability", "gpuCount": 1})
        return self._call("POST", "/pods", b)

    def create_cpu_pod(self, name, template_id, volume_id, env, ports, vcpu=2, flavors=("cpu3c", "cpu5c")):
        b = self._pod_body(name, template_id, volume_id, env, ports)
        b.update({"computeType": "CPU", "cpuFlavorIds": list(flavors), "cpuFlavorPriority": "availability", "vcpuCount": vcpu})
        return self._call("POST", "/pods", b)

    def get_pod(self, pod_id): return self._call("GET", f"/pods/{pod_id}")
    def terminate_pod(self, pod_id): self._call("DELETE", f"/pods/{pod_id}", ok404=True)
    def list_pods(self, name_prefix=""):
        pods = self._call("GET", "/pods") or []
        return [p for p in pods if p.get("name", "").startswith(name_prefix)]
    def get_volume(self, vid): return self._call("GET", f"/networkvolumes/{vid}")
    def create_volume(self, name, size_gb, dc="EU-RO-1"): return self._call("POST", "/networkvolumes", {"name": name, "size": size_gb, "dataCenterId": dc})
    def resize_volume(self, vid, size_gb): return self._call("PATCH", f"/networkvolumes/{vid}", {"size": size_gb})
    def delete_volume(self, vid): self._call("DELETE", f"/networkvolumes/{vid}", ok404=True)
    def save_template(self, name, image, ports, env, container_disk_gb=10, registry_auth_id=None, template_id=None):
        b = {"name": name, "imageName": image, "ports": ports, "env": env, "containerDiskInGb": container_disk_gb, "volumeInGb": 0, "isPublic": False}
        if registry_auth_id: b["containerRegistryAuthId"] = registry_auth_id
        return self._call("PATCH", f"/templates/{template_id}", b) if template_id else self._call("POST", "/templates", b)
    def billing_pods(self, since_iso, until_iso): return self._call("GET", f"/billing/pods?startTime={since_iso}&endTime={until_iso}") or []
    def billing_volumes(self, since_iso, until_iso): return self._call("GET", f"/billing/networkvolumes?startTime={since_iso}&endTime={until_iso}") or []

def pod_public_endpoint(pod, private_port):
    ip = pod.get("publicIp")
    mapping = pod.get("portMappings") or {}
    public = mapping.get(str(private_port))
    if not ip or not public: raise RunPodError(0, f"pod {pod.get('id')} has no public endpoint for {private_port}")
    return ip, int(public)
```

```python
        self.id, self.command, self.env, self.cwd, self.log_path = task_id, command, env, cwd, log_path

    env = os.environ.copy(); hfs = os.environ.get("HFS", "")

class Registry:
    def __init__(self, slots, log_dir): self.slots, self.log_dir, self.tasks, self.lock = slots, log_dir, {}, threading.Lock()
    def busy(self): return sum(1 for t in self.tasks.values() if t.state == "running")
    def submit(self, body):
        with self.lock:
            if self.busy() >= self.slots: return None
            t = Task(body["task_id"], body["command"], body.get("env") or {}, body.get("cwd"), body.get("log_path"))
            self.tasks[t.id] = t
        threading.Thread(target=self._run, args=(t,), daemon=True).start(); return t
    def _run(self, t):
        os.makedirs(self.log_dir, exist_ok=True)
        path = t.log_path or os.path.join(self.log_dir, f"{t.id}.log")
        os.makedirs(os.path.dirname(path), exist_ok=True)

            t.proc.wait()
```

```python
def plan_packages(entries, max_bytes=int(1.5 * 2**30)):
    by_dir = {}
    for e in sorted(entries, key=lambda e: e.remote):
        by_dir.setdefault(os.path.dirname(e.remote), []).append(e)
    packages, cur, cur_size = [], [], 0
    for d in sorted(by_dir):
        for e in by_dir[d]:
            if e.size > max_bytes:
                if cur: packages.append(cur); cur, cur_size = [], 0
                packages.append([e]); continue
            if cur and cur_size + e.size > max_bytes:
                packages.append(cur); cur, cur_size = [], 0
            cur.append(e); cur_size += e.size
    if cur: packages.append(cur)
    return packages

def build_rclone_args(package, target, direction, local_root, remote_root, tmp_dir):
    files_from = os.path.join(tmp_dir, f"files_{uuid.uuid4().hex[:8]}.txt")
    with open(files_from, "w") as f:
        for e in package: f.write(os.path.relpath(e.local, local_root) + "\n")
    remote = f":sftp:{remote_root}"
    src, dst = (local_root, remote) if direction == "up" else (remote, local_root)
    return ["copy", src, dst, "--files-from", files_from, f"--sftp-host={target.host}", f"--sftp-port={target.port}",
            f"--sftp-user={target.user}", f"--sftp-key-file={target.key_path}", "--sftp-set-modtime",
```

```python
# rpfarm/deps.py
import os, re
from .sync import FileEntry
_SEQ = re.compile(r"\$F\d*|%0?\d*d|<UDIM>|<udim>|#{2,}|\$\(F\d*\)|\$\{F\d*\}")

def collect_refs():
    import hou
    out = [hou.hipFile.path()]
    for parm, path in hou.fileReferences():
        if not path or path.startswith("op:") or path.startswith("opdef:"): continue
        p = hou.text.expandString(path) if "$" in path else path
        if _SEQ.search(p): p = os.path.dirname(p)
        out.append(p)
    return out

def resolve_entries(paths, job_dir, remote_project):
    job_dir = os.path.normpath(job_dir); seen, entries, pmap = set(), [], {job_dir: remote_project}
    def add(fp):
        fp = os.path.normpath(fp)
        if fp in seen or not os.path.isfile(fp): return
        seen.add(fp)
        if fp == job_dir or fp.startswith(job_dir + os.sep):
            remote = f"{remote_project}/{os.path.relpath(fp, job_dir).replace(os.sep, '/')}"
        else:
            drive, rest = os.path.splitdrive(fp)
            remote = f"{remote_project}/_ext{(drive.rstrip(':') + rest).replace(os.sep, '/')}"
            root = os.path.dirname(fp); pmap.setdefault(root, f"{remote_project}/_ext{(drive.rstrip(':') + os.path.splitdrive(root)[1]).replace(os.sep, '/')}")
        entries.append(FileEntry(local=fp, remote=remote, size=os.path.getsize(fp)))
    for p in paths:
        if os.path.isdir(p):
            for r, _, fs in os.walk(p):
                for f in fs: add(os.path.join(r, f))
        else: add(p)
    return entries, pmap
```
